<script>
(function () {
  const rows = []
  const add = (k, v) => rows.push([k, String(v)])

  add("画面幅(CSS px)", window.innerWidth + " × " + window.innerHeight)
  add("解像度倍率", window.devicePixelRatio)

  // メインCSS（index-*.css）が実際に適用されているか
  let main = null, mainRules = "—"
  for (const s of document.styleSheets) {
    if (s.href && /index-[^/]*\.css/.test(s.href)) main = s
  }
  add("メインCSSのlink", main ? main.href.split("/").pop() : "見つからない")
  if (main) {
    try { mainRules = main.cssRules.length + " 件" } catch (e) { mainRules = "読めない(" + e.name + ")" }
  }
  add("メインCSSのルール数", mainRules)

  // 段階ごとの適用判定
  const body = document.getElementById("quartz-body")
  add("#quartz-bodyのgrid列", body ? getComputedStyle(body).gridTemplateColumns : "要素なし")
  add("↑これが none なら", "レイヤー内CSSが効いていない")

  // レイヤー外CSS（custom.scss由来）が効いているか。#term-panel は @layer の外にある
  const panel = document.getElementById("term-panel")
  add("レイヤー外CSSの判定", panel ? getComputedStyle(panel).gridArea : "要素なし")
  add("↑これが grid-sidebar-right 以外なら", "メインCSSが丸ごと届いていない")

  // ブラウザの対応状況
  add("@layer 対応", typeof CSSLayerBlockRule !== "undefined")
  add(":has() 対応", (window.CSS && CSS.supports && CSS.supports("selector(:has(*))")) || false)
  add("テーマ属性", document.documentElement.getAttribute("saved-theme") || "(未設定)")
  add("読み込んだCSSの数", document.styleSheets.length)

  // 自己修復（cssGuard）の履歴。崩れが起きた証拠はここに残る
  let guard = "記録なし（崩れは起きていない）"
  try {
    const log = JSON.parse(localStorage.getItem("cssGuardLog") || "[]")
    if (log.length) {
      guard = log
        .slice(0, 5)
        .map((r) => r.t.replace("T", " ").slice(0, 19) + " " + r.status + " (幅" + r.w + ")")
        .join(" / ")
    }
  } catch (e) {
    guard = "読めない(" + e.name + ")"
  }
  add("自己修復の履歴", guard)
  add("UA", navigator.userAgent)

  const el = document.getElementById("diag")
  el.innerHTML =
    '<div style="font-weight:700;font-size:17px;margin-bottom:8px">診断結果</div>' +
    rows
      .map(
        ([k, v]) =>
          '<div style="margin:2px 0"><span style="opacity:.7">' +
          k +
          "：</span><b style=\"word-break:break-all\">" +
          v +
          "</b></div>",
      )
      .join("")
})()
</script>

上の枠の内容をそのまま撮って送ってください。判定に使うのは次の4行です。

- **メインCSSのルール数** … `0 件`／「読めない」「見つからない」ならCSS自体が届いていません
- **#quartz-bodyのgrid列** … `none` ならレイアウトCSSだけが効いていません
- **@layer 対応** … `false` なら、このブラウザが `@layer` を理解できないことが原因です
- **自己修復の履歴** … `retry1` `recovered` があれば、その時刻に取得失敗が起きて自動で復旧したという記録です。`failed` は3回試しても復旧できなかったことを意味します

## 実測済みの事実（2026-08-05）

iPad（Safari 26.5.2・画面幅1366px）で、**メインCSSのルール数が `0 件`** になる状態を実機で確認。ファイル名は最新の `index-c13faa6a.css` を正しく指しており、サーバ側は同時刻に HTTP 200・38,488バイトを返していた。`@layer` 対応も `true`。よって原因は「ファイル名のズレ（404）」でも「ブラウザの `@layer` 非対応」でもなく、**その1ファイルの取得が端末側で失敗し、失敗結果がキャッシュされてリロードでも直らなくなる**事象。

対策として `Head.tsx` に自己修復スクリプト（cssGuard）を入れ、空シートを検知したら `?cssguard=` 付きで最大3回まで読み直すようにした。上の「自己修復の履歴」はその記録。
